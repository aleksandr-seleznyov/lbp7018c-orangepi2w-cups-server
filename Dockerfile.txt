FROM debian:bookworm-slim

# Build arguments
ARG DEBIAN_FRONTEND=noninteractive
ARG TZ="Europe/Amsterdam"

# Metadata
LABEL org.opencontainers.image.source="https://github.com/aleksandr-seleznyov/lbp7018c-orangepi2w-cups-server"
LABEL org.opencontainers.image.description="Optimized Canon CAPT Driver Container for LBP7018C"
LABEL org.opencontainers.image.version="3.0-optimized"

# Environment variables
ENV DEBIAN_FRONTEND=noninteractive \
    TZ=${TZ} \
    PRINTER_NAME=LBP7018C \
    DEVICE_PATH=/dev/usb/lp0

# Install ONLY what's needed for CAPT driver
# Removed: CUPS (running on host), avahi-daemon (not needed), printer-driver-* (not needed)
RUN apt-get update -qq && \
    apt-get install -y --no-install-recommends \
    # Package management and build essentials
    dpkg \
    ca-certificates \
    # Printing and rendering dependencies required by Canon packages
    cups \
    ghostscript \
    # Core dependencies for CAPT driver
    libc6:i386 \
    libstdc++6:i386 \
    libglade2-0:i386 \
    libatk1.0-0:i386 \
    libcairo2:i386 \
    libgtk2.0-0:i386 \
    libpango1.0-0:i386 \
    libjpeg62-turbo:i386 \
    libxml2:i386 \
    libpopt0:i386 \
    # CUPS client libraries
    libcups2:i386 \
    libcupsimage2:i386 \
    cups-client \
    cups-ppdc \
    # Utilities
    wget \
    usbutils \
    procps \
    && rm -rf /var/lib/apt/lists/*

# Download Canon CAPT driver v2.71
RUN wget https://gdlp01.c-wss.com/gds/6/0100004596/05/linux-capt-drv-v271-uken.tar.gz \
    --no-check-certificate \
    -O /tmp/linux-capt-drv.tar.gz

# Extract Canon CAPT driver
RUN tar -xzf /tmp/linux-capt-drv.tar.gz -C /tmp

# Install Canon CAPT common package
RUN echo "Installing cndrvcups-common..." && \
    dpkg -i /tmp/linux-capt-drv-v271-uken/32-bit_Driver/Debian/cndrvcups-common_3.21-1_i386.deb && \
    echo "cndrvcups-common installed successfully"

# Install Canon CAPT driver package
RUN echo "Installing cndrvcups-capt..." && \
    dpkg -i /tmp/linux-capt-drv-v271-uken/32-bit_Driver/Debian/cndrvcups-capt_2.71-1_i386.deb && \
    echo "cndrvcups-capt installed successfully"

# Cleanup temporary files
RUN rm -rf /tmp/linux-capt-drv* /tmp/*.tar.gz

# Create necessary directories
RUN mkdir -p /var/log/CCPD && \
    mkdir -p /var/ccpd && \
    chmod 755 /var/ccpd

# Configure CUPS minimally (only for ccpd validation, won't accept external jobs)
RUN sed -i 's/Listen localhost:631/Listen 127.0.0.1:59631/' /etc/cups/cupsd.conf && \
    sed -i 's/Browsing On/Browsing Off/' /etc/cups/cupsd.conf && \
    sed -i 's/<Location \/>/<Location \/>\n  Allow All/' /etc/cups/cupsd.conf && \
    sed -i 's/<Location \/admin>/<Location \/admin>\n  Allow All\n  Require user @SYSTEM/' /etc/cups/cupsd.conf && \
    sed -i 's/<Location \/admin\/conf>/<Location \/admin\/conf>\n  Allow All/' /etc/cups/cupsd.conf && \
    echo "DefaultEncryption Never" >> /etc/cups/cupsd.conf

# Create CUPS admin user for lpadmin commands
RUN useradd -r -G lpadmin cupsadmin && \
    echo "cupsadmin:cupsadmin" | chpasswd

# Configure ccpd with proper logging
RUN cat > /etc/ccpd.conf << 'EOF'
# Canon Printer Daemon Configuration - Optimized for Performance

<Path>
# CUPS configuration path
CUPS_ConfigPath   /etc/cups/

# Log directory with verbose logging
LogDirectoryPath /var/log/CCPD/
</Path>

# Printer configuration
<Printer LBP7018C>
DevicePath /dev/usb/lp0
</Printer>

<Ports>
# Status monitoring socket port
UI_Port  59787
PDATA_Port  59687
</Ports>

# Performance tuning
<Options>
# Reduce timeout values for faster response
Timeout 10
RetryCount 3
</Options>
EOF

# Copy minimal CUPS client configuration
RUN mkdir -p /etc/cups && \
    cat > /etc/cups/client.conf << 'EOF'
# CUPS client configuration
ServerName localhost
Encryption Never
EOF

# Create optimized entrypoint script
COPY entrypoint.sh /entrypoint.sh
RUN chmod +x /entrypoint.sh

# Expose ccpd ports (for communication with host CUPS)
EXPOSE 59787
EXPOSE 59687

# Health check
HEALTHCHECK --interval=30s --timeout=10s --start-period=40s --retries=3 \
    CMD ccpdadmin | grep -q "59787" || exit 1

CMD ["/entrypoint.sh"]
